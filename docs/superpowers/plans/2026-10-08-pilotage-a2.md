# Pilotage A2 — marchés historiques présentés à part Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Afficher sur `/pilotage` un bloc « Marchés historiques » (marchés sans opportunité liée, répartis par issue, sans taux), sans toucher au taux de succès ni au taux de perte, qui restent calculés sur les seuls lots.

**Architecture:** Une fonction pure `calculerMarchesHistoriques` dans `lib/pilotage/calculs.ts` (le statut prime sur le concurrent gagnant), alimentée par deux champs de plus lus par `getPilotageData` (`concurrentGagnant`, `dateDernierStatut`), un nouveau bloc `historiques` dans `PilotageData`, une carte `IndicateurCard` dans `PilotageClient`, et deux changements de la qualité des données (libellé, compteur d'incohérence).

**Tech Stack:** Next.js 15, React 19, Prisma 7, Playwright (tests unitaires sous `playwright.unit.config.ts`, E2E sous `tests/v1`).

**Spec:** `docs/superpowers/specs/2026-10-08-pilotage-a-design.md` (section « Tranche A2 », révisée le 2026-10-08).

## Clarification de conception (à confirmer par Abel)

La spec écrit « `ANNULE` ou `RESILIE` → perdu après attribution ». Or l'indicateur de conversion exclut un marché annulé **avant** attribution (règle du 2026-10-02 : une procédure arrêtée avant attribution est une annulation). Pour rester cohérent, ce plan classe `ANNULE` avec `attribueUnJour = false` en `SANS_SUITE`, et réserve `PERDU_APRES_ATTRIBUTION` à `RESILIE` et à `ANNULE` avec `attribueUnJour = true`.

// ── Marchés historiques (sans opportunité liée), présentés à part, sans taux ──────────

export type IssueHistorique = 'GAGNE' | 'PERDU_APRES_ATTRIBUTION' | 'SANS_SUITE' | 'A_QUALIFIER'

export interface LigneHistorique {
  id: string
  numero: string
  objet: string
  montant: number
  issue: IssueHistorique
  href: string
}

export interface ResultatMarchesHistoriques {
  total: Repartition
  parIssue: Record<IssueHistorique, Repartition>
  lignes: LigneHistorique[]
  exclusSansDate: ElementLie[]
}

const STATUTS_A_QUALIFIER = ['OFFRE_DEPOSEE', 'EN_ATTENTE_ATTRIBUTION', 'ATTRIBUE_PROVISOIREMENT']

export function estMarcheHistorique(m: MarchePilotage): boolean {
  return !m.aOpportunite && !STATUTS_AVANT_DEPOT.includes(m.statut)
}

/** Le statut prime sur le concurrent gagnant. Une annulation avant attribution est une procédure sans suite. */
export function issueMarcheHistorique(m: MarchePilotage): IssueHistorique | null {
  if ((STATUTS_ATTRIBUES as string[]).includes(m.statut)) return 'GAGNE'
  if (m.statut === 'RESILIE' || (m.statut === 'ANNULE' && m.attribueUnJour)) return 'PERDU_APRES_ATTRIBUTION'
  if (m.statut === 'INFRUCTUEUX' || m.statut === 'ANNULE') return 'SANS_SUITE'
  if (STATUTS_A_QUALIFIER.includes(m.statut)) return 'A_QUALIFIER'
  return null
}

const dateRattachement = (m: MarchePilotage): Date | null => m.dateAttribution ?? m.dateDernierStatut

export function calculerMarchesHistoriques(marches: MarchePilotage[], periode: Periode): ResultatMarchesHistoriques {
  const vide = (): Repartition => ({ nombre: 0, valeur: 0 })
  const parIssue: Record<IssueHistorique, Repartition> = {
    GAGNE: vide(), PERDU_APRES_ATTRIBUTION: vide(), SANS_SUITE: vide(), A_QUALIFIER: vide(),
  }
  const total = vide()
  const lignes: LigneHistorique[] = []
  const exclusSansDate: ElementLie[] = []

  for (const m of marches) {
    if (!estMarcheHistorique(m)) continue
    const issue = issueMarcheHistorique(m)
    if (issue === null) continue
    const date = dateRattachement(m)
    if (!date) { exclusSansDate.push(lienMarche(m)); continue }
    if (!dansPeriode(date, periode)) continue
    parIssue[issue].nombre++
    parIssue[issue].valeur += m.montant
    total.nombre++
    total.valeur += m.montant
    lignes.push({ id: m.id, numero: m.numero, objet: m.objet, montant: m.montant, issue, href: `/marches/${m.id}` })
  }
  return { total, parIssue, lignes, exclusSansDate }
}

- [ ] **Step 2: Calcul SQL indépendant, par issue, sur 12 mois glissants**

```sql
with h as (
  select m.id, m.statut::text as statut, m.montant,
    (m."dateAttributionDefinitive" is not null or m.statut::text = 'RESILIE'
      or m.statut::text in ('ATTRIBUE_DEFINITIVEMENT','EN_ATTENTE_LIVRAISON_OS','EN_EXECUTION','EXECUTE_ATTENTE_GARANTIES','CLOTURE')
      or exists (select 1 from historique_statuts hs where hs."marcheId" = m.id
        and hs."nouveauStatut"::text in ('ATTRIBUE_DEFINITIVEMENT','EN_ATTENTE_LIVRAISON_OS','EN_EXECUTION','EXECUTE_ATTENTE_GARANTIES','CLOTURE'))) as attribue,
    coalesce(m."dateAttributionDefinitive",
      (select min(hs."createdAt") from historique_statuts hs where hs."marcheId" = m.id
        and hs."nouveauStatut"::text in ('ATTRIBUE_DEFINITIVEMENT','EN_ATTENTE_LIVRAISON_OS','EN_EXECUTION','EXECUTE_ATTENTE_GARANTIES','CLOTURE')),
      case m.statut::text when 'INFRUCTUEUX' then m."dateInfructueux" when 'ANNULE' then m."dateAnnulation" when 'RESILIE' then m."dateResiliation" end,
      (select max(hs."createdAt") from historique_statuts hs where hs."marcheId" = m.id)) as d
  from marches m
  where m."opportuniteId" is null
    and not exists (select 1 from opportunites o where o."marcheId" = m.id)
    and m.statut::text not in ('OPPORTUNITE_IDENTIFIEE','DOSSIER_EN_PREPARATION')
)
select case
    when statut in ('ATTRIBUE_DEFINITIVEMENT','EN_ATTENTE_LIVRAISON_OS','EN_EXECUTION','EXECUTE_ATTENTE_GARANTIES','CLOTURE') then 'GAGNE'
    when statut = 'RESILIE' or (statut = 'ANNULE' and attribue) then 'PERDU_APRES_ATTRIBUTION'
    when statut in ('INFRUCTUEUX','ANNULE') then 'SANS_SUITE'
    when statut in ('OFFRE_DEPOSEE','EN_ATTENTE_ATTRIBUTION','ATTRIBUE_PROVISOIREMENT') then 'A_QUALIFIER' end as issue,
  count(*) as nombre, sum(montant) as valeur
from h
where d >= now() - interval '1 year' and d <= now()
group by 1 order by 1;
```
